01. Configurar el Proyecto para Producción

Preparación de Dockerfile multicapa, orquestador de producción y variables seguras en PC Local.

🚀 Paso 1: Crear Dockerfile de Producción para Next.js
[PC LOCAL] frontend/Dockerfile.prod

Construye una imagen ultraligera multicapa que compila Next.js en una etapa previa y entrega el runner final optimizado:

DOCKERFILE
FROM node:20-alpine AS deps
WORKDIR /app
RUN apk add --no-cache libc6-compat
COPY package.json package-lock.json* ./
RUN npm install --frozen-lockfile || npm install

FROM node:20-alpine AS builder
WORKDIR /app
COPY --from=deps /app/node_modules ./node_modules
COPY . .
ARG NEXT_PUBLIC_API_URL
ENV NEXT_PUBLIC_API_URL=$NEXT_PUBLIC_API_URL
ENV NODE_ENV=production
RUN npm run build

FROM node:20-alpine AS runner
WORKDIR /app
ENV NODE_ENV=production
ENV PORT=3000
ENV HOSTNAME="0.0.0.0"

RUN addgroup --system --gid 1001 nodejs && adduser --system --uid 1001 nextjs
COPY --from=builder /app/package.json ./
COPY --from=builder /app/node_modules ./node_modules
COPY --from=builder /app/.next ./.next
COPY --from=builder /app/public ./public

USER nextjs
EXPOSE 3000
CMD ["npm", "run", "start"]
✔ Verificación Observable
Comprueba la existencia del archivo: ls -lh frontend/Dockerfile.prod (tamaño aprox. 800 bytes).
🚀 Paso 2: Crear el Orquestador docker-compose.prod.yml
[PC LOCAL] docker-compose.prod.yml

Aísla PostgreSQL en la red interna de Docker y publica únicamente hacia 127.0.0.1:

YAML
services:
  postgres:
    image: postgres:16-alpine
    container_name: gestornotas_postgres
    restart: unless-stopped
    environment:
      POSTGRES_DB: ${POSTGRES_DB:-gestornotas_db}
      POSTGRES_USER: ${POSTGRES_USER:-gestornotas_user}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD}
    volumes:
      - postgres_data:/var/lib/postgresql/data
    networks:
      - gestornotas_network
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${POSTGRES_USER:-gestornotas_user} -d ${POSTGRES_DB:-gestornotas_db}"]
      interval: 10s
      timeout: 5s
      retries: 5

  backend:
    build:
      context: ./backend
      dockerfile: Dockerfile
    container_name: gestornotas_backend
    restart: unless-stopped
    command: >
      sh -c "python manage.py migrate --noinput &&
             python manage.py collectstatic --noinput &&
             python manage.py runserver 0.0.0.0:8000"
    volumes:
      - media_volume:/app/media
      - static_volume:/app/staticfiles
    ports:
      - "127.0.0.1:8000:8000"
    environment:
      DJANGO_SECRET_KEY: ${DJANGO_SECRET_KEY}
      DJANGO_DEBUG: "False"
      DJANGO_ALLOWED_HOSTS: ${DJANGO_ALLOWED_HOSTS}
      USE_SQLITE: "False"
      POSTGRES_DB: ${POSTGRES_DB}
      POSTGRES_USER: ${POSTGRES_USER}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD}
      POSTGRES_HOST: postgres
      POSTGRES_PORT: 5432
      GEMINI_API_KEY: ${GEMINI_API_KEY}
      CORS_ALLOWED_ORIGINS: ${CORS_ALLOWED_ORIGINS}
    depends_on:
      postgres:
        condition: service_healthy
    networks:
      - gestornotas_network

  web:
    build:
      context: ./frontend
      dockerfile: Dockerfile.prod
      args:
        NEXT_PUBLIC_API_URL: ${NEXT_PUBLIC_API_URL}
    container_name: gestornotas_frontend
    restart: unless-stopped
    ports:
      - "127.0.0.1:3000:3000"
    environment:
      NEXT_PUBLIC_API_URL: ${NEXT_PUBLIC_API_URL}
      PORT: 3000
    depends_on:
      - backend
    networks:
      - gestornotas_network

volumes:
  postgres_data:
    driver: local
  media_volume:
    driver: local
  static_volume:
    driver: local

networks:
  gestornotas_network:
    driver: bridge

02. Crear Infraestructura en Google Cloud

Creación de máquina virtual e2-medium, IP estática y reglas de Firewall en GCP.

🚀 Paso 1: Reservar Dirección IPv4 Externa Estática
[PC LOCAL / CLI]

Reserva una IP pública inmutable en us-central1 para que no cambie al apagar o reiniciar la VM:

BASH
# Reservar IP estática
gcloud compute addresses create gestornotas-ip-estatica --region=us-central1

# Consultar la IP obtenida
export STATIC_IP=$(gcloud compute addresses describe gestornotas-ip-estatica --region=us-central1 --format='value(address)')
echo "Tu IP Estática es: $STATIC_IP"
🚀 Paso 2: Crear Instancia Compute Engine e2-medium
[PC LOCAL / CLI]

Aprovisiona la VM con 2 vCPU, 4 GB RAM, 30 GB de disco equilibrado y Ubuntu 22.04 LTS:

BASH
gcloud compute instances create gestornotas-vm-prod \
    --zone=us-central1-a \
    --machine-type=e2-medium \
    --network-interface=address=$STATIC_IP,network-tier=PREMIUM \
    --image-family=ubuntu-2204-lts \
    --image-project=ubuntu-os-cloud \
    --boot-disk-size=30GB \
    --boot-disk-type=pd-balanced \
    --tags=http-server,https-server
✔ Verificación Observable
gcloud compute instances describe gestornotas-vm-prod --zone=us-central1-a --format='value(status)' → Salida: RUNNING.
🚀 Paso 3: Conectar por SSH al Servidor
[PC LOCAL]
BASH
gcloud compute ssh gestornotas-vm-prod --zone=us-central1-a

03. Instalar Herramientas en el Servidor

Aprovisionamiento de Docker, Docker Compose, 2 GB Swap, UFW y Nginx en Ubuntu.

🚀 Paso 1: Actualizar Paquetes y Configurar 2 GB Swap
[SSH / SERVIDOR]
BASH
sudo apt update && sudo apt upgrade -y
sudo apt install -y curl wget git htop ufw software-properties-common ca-certificates apt-transport-https

# Swap de 2 GB
sudo fallocate -l 2G /swapfile
sudo chmod 600 /swapfile
sudo mkswap /swapfile
sudo swapon /swapfile
echo '/swapfile none swap sw 0 0' | sudo tee -a /etc/fstab
sudo sysctl vm.swappiness=10
echo 'vm.swappiness=10' | sudo tee -a /etc/sysctl.conf

free -h
🚀 Paso 2: Instalar Docker Engine y Docker Compose Oficial
[SSH / SERVIDOR]
BASH
curl -fsSL https://get.docker.com -o get-docker.sh
sudo sh get-docker.sh
sudo usermod -aG docker $USER
sudo systemctl enable docker
sudo systemctl start docker
rm get-docker.sh
newgrp docker

# Probar ejecución no-root
docker run --rm hello-world
🚀 Paso 3: Instalar Nginx y Activar Firewall UFW
[SSH / SERVIDOR]
BASH
sudo apt install -y nginx
sudo systemctl enable nginx
sudo systemctl start nginx

sudo ufw allow 22/tcp comment 'SSH Admin'
sudo ufw allow 80/tcp comment 'HTTP Nginx'
sudo ufw allow 443/tcp comment 'HTTPS Nginx'
sudo ufw --force enable
sudo ufw status verbose

NGINX
server {
    listen 80;
    server_name <TU_IP_PUBLICA_GCP> _;

    client_max_body_size 50M;

    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;

    location /api/ {
        proxy_pass http://127.0.0.1:8000/api/;
        proxy_read_timeout 180s;
        proxy_connect_timeout 180s;
    }

    location /admin/ {
        proxy_pass http://127.0.0.1:8000/admin/;
    }

    location /media/ {
        proxy_pass http://127.0.0.1:8000/media/;
    }

    location /static/ {
        proxy_pass http://127.0.0.1:8000/static/;
    }

    location / {
        proxy_pass http://127.0.0.1:3000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
    }
}
BASH
sudo rm -f /etc/nginx/sites-enabled/default

05. Actualizar Producción (Zero-Downtime)

Procedimiento seguro de CI/CD manual con backups de base de datos y rollback.

🚀 Pipeline de Actualización Segura en Servidor
[SSH / SERVIDOR]
BASH
cd ~/proyecto-gestor-notas

# 1. Backup de seguridad antes de tocar nada
mkdir -p ~/backups_db
docker compose -f docker-compose.prod.yml exec postgres pg_dump -U gestornotas_admin gestornotas_prod_db | gzip > ~/backups_db/backup_pre_update_$(date +%Y%m%d_%H%M%S).sql.gz

# 2. Descargar código
git pull origin main

# 3. Reconstruir y levantar contenedores en caliente
docker compose -f docker-compose.prod.yml build
docker compose -f docker-compose.prod.yml up -d --remove-orphans

# 4. Migraciones y estáticos
docker compose -f docker-compose.prod.yml exec backend python manage.py migrate --noinput
docker compose -f docker-compose.prod.yml exec backend python manage.py collectstatic --noinput

# 5. Limpieza de imágenes huérfanas
docker image prune -f

# 6. Verificación
curl -s -I http://localhost | head -n 3
⚠️ Procedimiento de Reversión (Rollback Inmediato)
[SSH / SERVIDOR]
BASH
git reset --hard <HASH_COMMIT_ANTERIOR>
gunzip -c ~/backups_db/backup_pre_update_YYYYMMDD_HHMMSS.sql.gz | docker compose -f docker-compose.prod.yml exec -T postgres psql -U gestornotas_admin -d gestornotas_prod_db
docker compose -f docker-compose.prod.yml up -d --build --force-recreate
